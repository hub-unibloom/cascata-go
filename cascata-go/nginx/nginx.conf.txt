
map $http_upgrade $connection_upgrade {
    default upgrade;
    '' close;
}

# ═══════════════════════════════════════════════════════════════════════════════
# LAYER 0: EDGE DEFENSE (Nginx) - Defesa de borda antes do Go
# ═══════════════════════════════════════════════════════════════════════════════

# Rate limiting por IP (DDoS protection)
limit_req_zone $binary_remote_addr zone=edge_ddos:50m rate=5000r/s;
limit_req_zone $binary_remote_addr zone=api_limit:10m rate=1000r/s;
limit_req_zone $binary_remote_addr zone=auth_limit:10m rate=100r/s;

# Connection limiting (anti-slowloris)
limit_conn_zone $binary_remote_addr zone=conn_limit:10m;
limit_conn_zone $server_name zone=server_conn:10m;

# Map para extrair tenant do server_name (subdomínio)
map $host $tenant_slug {
    default "";
    # Extrai subdomain de tenant.cascata.io ou tenant.custom.com
    ~^(?<subdomain>[a-z0-9_-]+)\.(?<domain>.+)$ $subdomain;
    # IPs e localhost não têm tenant
    ~^\d+\.\d+\.\d+\.\d+$ "";
    ~^localhost$ "";
}

# Map para verificar se tem User-Agent (bloqueia bots maliciosos)
map $http_user_agent $has_user_agent {
    default 1;
    "" 0;
    "-" 0;
}

# ============================================================
# CASCATA ZERO-NETWORK HOT-PATH (Hyper-Cluster)
# Upstream balanceia a carga usando Unix Domain Sockets 
# direto na memória RAM, ignorando a pilha TCP/IP do Linux.
# ============================================================
upstream backend_data_sockets {
    # Cascata Hyper-Cluster: 3 Workers (control=1, data=2, engine=3)
    # O NGINX automaticamente ignora workers que estiverem down
    server unix:/tmp/cascata_sockets/worker_1.sock max_fails=3 fail_timeout=5s;
    server unix:/tmp/cascata_sockets/worker_2.sock max_fails=3 fail_timeout=5s;
    server unix:/tmp/cascata_sockets/worker_3.sock max_fails=3 fail_timeout=5s;
    
    # Fallback de segurança para porta TCP caso sockets falhem
    server backend_data:3000 backup;
}

# --- BLOCO PADRÃO HTTP (Porta 80) ---
server {
    listen 80 default_server;
    server_name _;
    
    server_tokens off;
    
    # HARDENING: Increased to 5GB for Full Database Imports/Exports
    client_max_body_size 5G;

    location /.well-known/acme-challenge/ {
        root /var/www/html;
        allow all;
    }

    # Lógica de Isolamento Zero Trust
    location / {
        if ($host !~* ^(localhost|127\.0\.0\.1|172\.|10\.|[0-9]+\.[0-9]+\.[0-9]+\.[0-9]+)$ ) {
            return 444; 
        }

        limit_req zone=api_limit burst=20 nodelay;
        limit_conn conn_limit 20;
        
        proxy_pass http://frontend;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }

    # API Control Plane (Long Timeout for Restores)
    location /api/control/ {
        if ($host !~* ^(localhost|127\.0\.0\.1|172\.|10\.|[0-9]+\.[0-9]+\.[0-9]+\.[0-9]+)$ ) {
            return 444;
        }
        limit_req zone=api_limit burst=10 nodelay;
        # Control Plane permanece na porta TCP padrão (não focado em hiper-rendimento, mas estabilidade)
        proxy_pass http://backend_control:3000;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        
        # Extended timeouts for Import/Export operations
        proxy_read_timeout 3600s;
        proxy_send_timeout 3600s;
    }

    # Sovereign Elite: Status & Unseal (Always to Control Plane)
    location ~ ^/auth/sovereign/ {
        proxy_pass http://backend_control:3000;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }

    # API Data Plane (Hyper-Cluster Via Unix Sockets)
    location ~ ^/(api/data/.*|rpc/.*|auth/.*|storage/.*|edge/.*|tables/.*|rest/.*|vector/.*) {
        # ═════════════════════════════════════════════════════════════════
        # LAYER 0: EDGE DEFENSE - Primeira linha de defesa no Nginx
        # ═════════════════════════════════════════════════════════════════
        
        # 1. Bloqueia requests sem User-Agent (bots maliciosos)
        if ($has_user_agent = 0) {
            return 444; # Close connection without response
        }
        
        # 2. Rate limiting agressivo contra DDoS bruto (Layer 0)
        # Se exceder 5000 req/s por IP, descarta na borda (zero custo Go/Dragonfly)
        limit_req zone=edge_ddos burst=1000 nodelay;
        
        # 3. Connection limiting (anti-slowloris)
        limit_conn conn_limit 50;
        limit_conn server_conn 1000; # Limite por server_name
        
        # 4. Refinação: rate limit específico por tipo de rota
        limit_req zone=api_limit burst=50 nodelay;
        
        # ═════════════════════════════════════════════════════════════════
        # INJEÇÃO DE TENANT SLUG (para Go identificar sem parsear URL)
        # ═════════════════════════════════════════════════════════════════
        # Prioridade: 1. Header j existente, 2. Subdomínio extraído, 3. Vazio
        proxy_set_header X-Tenant-Slug $tenant_slug;
        
        # O Pulo do Gato: NGINX entrega direto na RAM via Socket
        proxy_pass http://backend_data_sockets;
        
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        
        proxy_buffering off;
        proxy_cache off;
        proxy_read_timeout 86400s;
        proxy_send_timeout 86400s;
    }
}

# --- BLOCO PADRÃO HTTPS (Porta 443) ---
server {
    listen 443 ssl default_server;
    server_name _;
    
    server_tokens off;
    client_max_body_size 5G; # Match HTTP limit

    ssl_certificate /etc/letsencrypt/live/system/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/system/privkey.pem;

    ssl_protocols TLSv1.2 TLSv1.3;
    ssl_ciphers HIGH:!aNULL:!MD5;
    ssl_prefer_server_ciphers on;

    # Session Hardening
    ssl_session_timeout 1d;
    ssl_session_cache shared:SSL:10m;
    ssl_session_tickets off;

    # HSTS (Enterprise Standard)
    add_header Strict-Transport-Security "max-age=15768000" always;
    add_header X-Frame-Options DENY;
    add_header X-Content-Type-Options nosniff;
    add_header X-XSS-Protection "1; mode=block";

    location / {
        return 444;
    }
}

include /etc/nginx/conf.d/dynamic/*.conf;
